function sendFloatingChatMessage() { let input = document.getElementById('floating-chat-input'); let text = input.value.trim(); const user = getCurrentUser(); if(text && user) { chatMessages.push({ id: Date.now(), sender: 'user', identifier: user.identifier, text: text, time: new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' }) }); saveData(); input.value = ''; renderFloatingChatMessages(); } }
