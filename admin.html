কোনো ক্লায়েন্ট মেসেজ পাঠায়নি।

`; } else { chatContainer.innerHTML = userChats.slice().reverse().map(c => `
ক্লায়েন্ট আইডি: ${c.identifier || 'ইউজার'} ${c.time || 'সাম্প্রতিক'}

${c.text}

`).join(''); } } // ৩. ডিপোজিট রিকোয়েস্ট লিস্ট let depContainer = document.getElementById('admin-deposits-list'); if(depContainer) { if(depositRequests.length === 0) { depContainer.innerHTML = `

কোনো ডিপোজিট রিকোয়েস্ট নেই।

`; } else { depContainer.innerHTML = depositRequests.slice().reverse().map(d => `
${d.identifier}${d.amount} ৳ (${d.planName})
মেথড: ${d.method} | TrxID: ${d.trxId}
${d.status} ${d.status === 'Pending' ? `` : ''}
`).join(''); } } // ৪. উইথড্র রিকোয়েস্ট লিস্ট let wdContainer = document.getElementById('admin-withdraws-list'); if(wdContainer) { if(withdrawRequests.length === 0) { wdContainer.innerHTML = `

কোনো উইথড্র রিকোয়েস্ট নেই।

`; } else { wdContainer.innerHTML = withdrawRequests.slice().reverse().map(w => `
${w.identifier}${w.amount} ৳
অ্যাকাউন্ট: ${w.number} (${w.method})
${w.status} ${w.status === 'Pending' ? `` : ''}
`).join(''); } } // ৫. ইউজার লিস্ট, পিন রিসেট ও ব্যালেন্স ম্যানেজমেন্ট (সম্পূর্ণ ক্লিকযোগ্য ও আলাদা বক্স) let usersContainer = document.getElementById('admin-users-list'); if(usersContainer) { if(usersList.length === 0) { usersContainer.innerHTML = `

কোনো রেজিস্টার্ড ইউজার নেই।

`; } else { usersContainer.innerHTML = usersList.map((u, index) => `
${u.identifier} উইথড্র পিন: ${u.transactionPin || 'নেই'}
মূল: ${u.balance.toFixed(2)} ৳ বোনাস: ${(u.bonusBalance || 0).toFixed(2)} ৳
`).join(''); } } } function updateGlobalNotice() { let val = document.getElementById('admin-global-notice-input').value.trim(); if(val) { globalNotice = val; saveAdminData(); alert("গ্লোবাল নোটিশ সফলভাবে আপডেট করা হয়েছে!"); } } function sendAdminReply(identifier) { let input = document.getElementById(`admin-reply-${identifier}`); // fallback if ID differs let text = input ? input.value.trim() : ""; if(text) { chatMessages.push({ sender: 'admin', identifier: identifier, text: text, time: new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' }) }); saveAdminData(); alert("উত্তর পাঠানো হয়েছে!"); renderAdminPanel(); } } function approveDeposit(id) { let dep = depositRequests.find(d => d.id === id); if(dep && dep.status === 'Pending') { dep.status = 'Success'; let u = usersList.find(user => user.identifier === dep.identifier); if(u) { let totalAdd = dep.amount * 2; // ১০০% ইনস্ট্যান্ট বোনাসসহ দ্বিগুণ যোগ u.balance += totalAdd; u.recharge = (u.recharge || 0) + dep.amount; u.notifications = u.notifications || []; u.notifications.unshift({ id: Date.now(), title: "ডিপোজিট সফল ও ১০০% বোনাস!", desc: `${dep.amount} ৳ ডিপোজিট এপ্রুভ হয়েছে এবং ১০০% বোনাসসহ মোট ${totalAdd} ৳ আপনার মূল ব্যালেন্সে যোগ হয়েছে।`, time: new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' }), read: false }); } saveAdminData(); alert("ডিপোজিট সফলভাবে এপ্রুভ করা হয়েছে!"); renderAdminPanel(); } } function approveWithdraw(id) { let wd = withdrawRequests.find(w => w.id === id); if(wd && wd.status === 'Pending') { wd.status = 'Success'; let u = usersList.find(user => user.identifier === wd.identifier); if(u) { u.notifications = u.notifications || []; u.notifications.unshift({ id: Date.now(), title: "উইথড্র সফল!", desc: `আপনার ${wd.amount} ৳ উইথড্র রিকোয়েস্ট সফলভাবে পরিশোধ করা হয়েছে।`, time: new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' }), read: false }); } saveAdminData(); alert("উইথড্র পেমেন্ট সফল বলে গণ্য করা হয়েছে!"); renderAdminPanel(); } } // ক্লিকযোগ্য পিন রিসেট ফাংশন function adminResetPin(index) { let u = usersList[index]; let newPin = prompt(`(${u.identifier}) এর নতুন ৪ ডিজিটের উইথড্র পিন দিন:`, "1234"); if(newPin && newPin.length === 4) { u.transactionPin = newPin; saveAdminData(); alert("উইথড্র পিন সফলভাবে পরিবর্তন করা হয়েছে!"); renderAdminPanel(); } else if(newPin) { alert("অবশ্যই ৪ ডিজিটের পিন দিতে হবে!"); } } // ক্লিকযোগ্য ব্যালেন্স যোগ ফাংশন function adminAddBalance(index) { let u = usersList[index]; let amt = prompt(`(${u.identifier}) এর অ্যাকাউন্টে কত টাকা যোগ করতে চান?`, "100"); if(amt && !isNaN(amt)) { u.balance += parseFloat(amt); u.notifications = u.notifications || []; u.notifications.unshift({ id: Date.now(), title: "ব্যালেন্স আপডেট", desc: `এডমিন কর্তৃক আপনার অ্যাকাউন্টে ${amt} ৳ যোগ করা হয়েছে।`, time: new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' }), read: false }); saveAdminData(); alert(`${amt} ৳ সফলভাবে যোগ করা হয়েছে!`); renderAdminPanel(); } } // ক্লিকযোগ্য ব্যালেন্স কর্তন ফাংশন function adminDeductBalance(index) { let u = usersList[index]; let amt = prompt(`(${u.identifier}) এর অ্যাকাউন্ট থেকে কত টাকা কাটতে চান?`, "50"); if(amt && !isNaN(amt)) { u.balance = Math.max(0, u.balance - parseFloat(amt)); u.notifications = u.notifications || []; u.notifications.unshift({ id: Date.now(), title: "ব্যালেন্স কর্তন", desc: `এডমিন কর্তৃক আপনার অ্যাকাউন্ট থেকে ${amt} ৳ কর্তন করা হয়েছে।`, time: new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' }), read: false }); saveAdminData(); alert(`${amt} ৳ কর্তন করা হয়েছে!`); renderAdminPanel(); } } function adminLogout() { localStorage.removeItem('taskbd_admin_logged'); window.location.href = 'index.html'; } renderAdminPanel();
